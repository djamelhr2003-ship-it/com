function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var prodSheet = ss.getSheetByName("اضافة منتجات"); var stockSheet = ss.getSheetByName("المخزون"); var shippingSheet = ss.getSheetByName("التوصيل"); var products = []; var stockMap = {}; var shippingRates = {}; if (shippingSheet && shippingSheet.getLastRow() > 1) { var shipRows = shippingSheet.getDataRange().getValues(); for (var s = 1; s < shipRows.length; s++) { var wilaya = shipRows[s][0]; var officePrice = Number(shipRows[s][1]) || 400; var homePrice = Number(shipRows[s][2]) || 700; if (wilaya && wilaya.toString().trim() !== "") { shippingRates[wilaya.toString().trim()] = { office: officePrice, home: homePrice }; } } } if (stockSheet && stockSheet.getLastRow() > 1) { var stockRows = stockSheet.getDataRange().getValues(); for (var j = 1; j < stockRows.length; j++) { var itemName = stockRows[j][0]; var itemQty = stockRows[j][5]; if (itemName && itemName.toString().trim() !== "Vendor" && itemName.toString().trim() !== "الصنف") { stockMap[itemName.toString().trim().toLowerCase()] = Number(itemQty) || 0; } } } if (prodSheet && prodSheet.getLastRow() > 1) { var prodRows = prodSheet.getDataRange().getValues(); for (var i = 1; i < prodRows.length; i++) { var name = prodRows[i][0]; var priceCell = prodRows[i][3]; // العمود D المخصص للسعر بدقة var customDesc = prodRows[i][2]; var imgUrl = ""; for (var c = 4; c < prodRows[i].length; c++) { var val = prodRows[i][c]; if (val && typeof val === "string" && (val.startsWith("http://") || val.startsWith("https://"))) { imgUrl = val; break; } } if (name && name.toString().trim() !== "" && name.toString().trim() !== "Feature" && name.toString().trim() !== "المنتج") { var cleanName = name.toString().trim(); var lookupKey = cleanName.toLowerCase(); var currentStock = stockMap[lookupKey] !== undefined ? stockMap[lookupKey] : 10; // استخراج السعر الحقيقي وتفادي النصوص العشوائية var cleanPrice = Number(String(priceCell).replace(/[^0-9.]/g, '')) || 0; var aiGeneratedDesc = customDesc && customDesc !== "" ? customDesc : `🌟 ${cleanName}: منتج أصلي عالي الجودة بتصميم عصري وأداء ممتاز يلبي احتياجاتك مع التوصيل السريع لجميع الولايات والدفع عند الاستلام.`; var finalImage = (imgUrl !== "") ? imgUrl : "https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=500"; products.push({ id: i, name: cleanName, price: cleanPrice, desc: aiGeneratedDesc, image: finalImage, stock: currentStock }); } } } return ContentService.createTextOutput(JSON.stringify({products: products, shipping: shippingRates})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({error: error.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("استقبال طلبات"); if (!sheet) sheet = ss.getActiveSheet(); var data = JSON.parse(e.postData.contents); var cartDetails = ""; if (data.cartItems && Array.isArray(data.cartItems)) { cartDetails = " [ITEMS:" + JSON.stringify(data.cartItems) + "]"; } sheet.appendRow([ "P0", "Order-" + Math.floor(Math.random() * 9000 + 1000), data.product || "منتج متجر DZ", "Received", data.date || new Date().toLocaleString(), data.totalPrice + " دج", data.delivery + " | الولاية: " + data.clientState + " | العنوان: " + data.address + cartDetails, data.name, data.phone ]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function onEdit(e) { try { var range = e.range; var sheet = range.getSheet(); if (sheet.getName() === "استقبال طلبات" && range.getColumn() === 4) { var statusValue = range.getValue().toString().trim().toLowerCase(); if (statusValue === "accepted" || statusValue === "مؤكد" || statusValue === "approved") { var row = range.getRow(); var rowValues = sheet.getRange(row, 1, 1, sheet.getLastColumn()).getValues()[0]; var itemsJson = ""; for (var k = 0; k < rowValues.length; k++) { var valStr = rowValues[k].toString(); if (valStr.indexOf("[ITEMS:") !== -1) { var match = valStr.match(/\[ITEMS:((.*?)?)\]/); if (match && match[1]) { itemsJson = match[1]; break; } } } if (itemsJson !== "") { var items = JSON.parse(itemsJson); var ss = SpreadsheetApp.getActiveSpreadsheet(); var stockSheet = ss.getSheetByName("المخزون"); if (stockSheet) { var stockRows = stockSheet.getDataRange().getValues(); items.forEach(function(cartItem) { var cartName = cartItem.name.toString().trim().toLowerCase(); var orderedQty = Number(cartItem.qty) || 1; for (var j = 1; j < stockRows.length; j++) { var stockItemName = stockRows[j][0] ? stockRows[j][0].toString().trim().toLowerCase() : ""; if (stockItemName === cartName) { var currentQty = Number(stockRows[j][5]) || 0; var newQty = Math.max(0, currentQty - orderedQty); stockSheet.getRange(j + 1, 6).setValue(newQty); break; } } }); sheet.getRange(row, 4).setValue(range.getValue() + " ✅ (تم خصم المخزون)"); } } } } } catch(err) { console.log("خطأ في الخصم التلقائي: " + err.toString()); } }
